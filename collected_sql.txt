CREATE OR REPLACE PACKAGE BODY pkg_web_api AS
    PROCEDURE has_main_admin(p_exists OUT NUMBER) IS
    BEGIN
        SELECT CASE WHEN COUNT(*)=0 THEN 0 ELSE 1 END
          INTO p_exists FROM app_users WHERE role='MAIN_ADMIN';
    END;

    PROCEDURE get_login(p_username IN VARCHAR2, p_user_id OUT NUMBER,
        p_password_hash OUT VARCHAR2, p_email OUT VARCHAR2,
        p_role OUT VARCHAR2, p_is_active OUT CHAR) IS
    BEGIN
        p_user_id := NULL; p_password_hash := NULL; p_email := NULL;
        p_role := NULL; p_is_active := 'N';
        SELECT user_id, password_hash, email, role, is_active
          INTO p_user_id, p_password_hash, p_email, p_role, p_is_active
          FROM app_users WHERE LOWER(username)=LOWER(TRIM(p_username));
    EXCEPTION WHEN NO_DATA_FOUND THEN NULL;
    END;

    PROCEDURE register_user(p_username IN VARCHAR2, p_password_hash IN VARCHAR2,
        p_email IN VARCHAR2, p_first_name IN VARCHAR2, p_last_name IN VARCHAR2,
        p_city IN VARCHAR2, p_user_id OUT NUMBER) IS
        l_fan_id NUMBER;
    BEGIN
        INSERT INTO dim_fan(first_name,last_name,email,city)
        VALUES(TRIM(p_first_name),TRIM(p_last_name),LOWER(TRIM(p_email)),p_city)
        RETURNING fan_id INTO l_fan_id;
        pkg_security_api.create_app_user(TRIM(p_username),p_password_hash,
            LOWER(TRIM(p_email)),'USER',l_fan_id,p_user_id);
    END;

    PROCEDURE bootstrap_main_admin(p_username IN VARCHAR2, p_password_hash IN VARCHAR2,
        p_email IN VARCHAR2, p_first_name IN VARCHAR2, p_last_name IN VARCHAR2,
        p_city IN VARCHAR2, p_user_id OUT NUMBER) IS
        l_count NUMBER;
        l_fan_id NUMBER;
    BEGIN
        SELECT COUNT(*) INTO l_count FROM app_users WHERE role='MAIN_ADMIN';
        IF l_count > 0 THEN
            RAISE_APPLICATION_ERROR(-20110,'Главный администратор уже создан.');
        END IF;
        INSERT INTO dim_fan(first_name,last_name,email,city)
        VALUES(TRIM(p_first_name),TRIM(p_last_name),LOWER(TRIM(p_email)),p_city)
        RETURNING fan_id INTO l_fan_id;
        pkg_security_api.create_app_user(TRIM(p_username),p_password_hash,
            LOWER(TRIM(p_email)),'MAIN_ADMIN',l_fan_id,p_user_id);
    END;

    PROCEDURE dashboard(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT (SELECT COUNT(*) FROM dim_player WHERE is_active='Y') AS players_count,
                   (SELECT COUNT(*) FROM dim_match WHERE match_status='SCHEDULED'
                      AND match_date>=TRUNC(SYSDATE)
                      AND opponent_name NOT LIKE 'TEST PERF %') AS upcoming_matches,
                   (SELECT COUNT(*) FROM tickets WHERE status='SOLD') AS sold_tickets,
                   (SELECT NVL(SUM(price),0) FROM tickets WHERE status='SOLD') AS ticket_revenue,
                   (SELECT COUNT(*) FROM app_users WHERE is_active='Y') AS active_users
              FROM dual;
    END;

    PROCEDURE list_players(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT p.player_id,p.first_name,p.last_name,p.position,p.jersey_number,
                   p.birth_date,p.country,p.height_cm,p.weight_kg,
                   NVL(f.fans_favorited,0) AS fans_favorited,
                   NVL(s.goals_total,0) AS goals_total,
                   NVL(s.assists_total,0) AS assists_total,
                   NVL(s.points_total,0) AS points_total
              FROM dim_player p
              LEFT JOIN v_player_popularity f ON f.player_id=p.player_id
              LEFT JOIN v_player_performance s ON s.player_id=p.player_id
             WHERE p.is_active='Y'
             ORDER BY p.last_name,p.first_name;
    END;

    PROCEDURE list_matches(p_date_from IN DATE, p_date_to IN DATE,
        p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT m.match_id,m.match_date,m.opponent_name,m.stadium_id,
                   st.name AS stadium,st.city,m.is_home,m.goals_for,m.goals_against,
                   m.match_status,
                   (SELECT COUNT(*) FROM tickets t WHERE t.match_id=m.match_id
                      AND (t.status IN ('FREE','RETURNED') OR (t.status='RESERVED' AND t.reserved_until<=SYSDATE)))
                      AS available_tickets
              FROM dim_match m JOIN dim_stadium st ON st.stadium_id=m.stadium_id
             WHERE m.opponent_name NOT LIKE 'TEST PERF %'
               AND (p_date_from IS NULL OR m.match_date>=TRUNC(p_date_from))
               AND (p_date_to IS NULL OR m.match_date<TRUNC(p_date_to)+1)
             ORDER BY m.match_date;
    END;

    PROCEDURE match_detail(p_match_id IN NUMBER, p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT m.match_id,m.match_date,m.opponent_name,m.stadium_id,
                   st.name AS stadium,st.city,st.capacity,m.is_home,
                   m.goals_for,m.goals_against,m.match_status
              FROM dim_match m JOIN dim_stadium st ON st.stadium_id=m.stadium_id
             WHERE m.match_id=p_match_id AND m.opponent_name NOT LIKE 'TEST PERF %';
    END;

    PROCEDURE list_merch(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT merch_id,name,category,price,description
              FROM dim_merch WHERE is_active='Y' ORDER BY category,name;
    END;

    PROCEDURE list_stadiums(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR SELECT stadium_id,name,city,capacity FROM dim_stadium ORDER BY name;
    END;

    PROCEDURE list_sectors(p_stadium_id IN NUMBER, p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT sector_id,stadium_id,sector_code,category,capacity
              FROM dim_sector WHERE stadium_id=p_stadium_id ORDER BY sector_code;
    END;

    PROCEDURE list_sector_rows(p_match_id IN NUMBER, p_sector_id IN NUMBER,
        p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT seat_row,COUNT(*) AS seats_total,
                   SUM(CASE WHEN status IN ('FREE','RETURNED') OR
                      (status='RESERVED' AND reserved_until<=SYSDATE) THEN 1 ELSE 0 END) AS seats_free
              FROM tickets WHERE match_id=p_match_id AND sector_id=p_sector_id
             GROUP BY seat_row ORDER BY seat_row;
    END;

    PROCEDURE list_seats(p_match_id IN NUMBER, p_sector_id IN NUMBER,
        p_seat_row IN NUMBER, p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT ticket_id,match_id,sector_id,seat_row,seat_number,price,
                   CASE WHEN status='RESERVED' AND reserved_until<=SYSDATE
                        THEN 'FREE' ELSE status END AS status,
                   CASE WHEN status='RESERVED' AND reserved_until>SYSDATE
                        THEN reserved_until END AS reserved_until
              FROM tickets
             WHERE match_id=p_match_id AND sector_id=p_sector_id AND seat_row=p_seat_row
             ORDER BY seat_number;
    END;

    PROCEDURE reserve_ticket(p_ticket_id IN NUMBER, p_app_user_id IN NUMBER,
        p_minutes IN NUMBER DEFAULT 15) IS
        l_fan_id NUMBER; l_status VARCHAR2(10); l_until DATE;
    BEGIN
        IF p_minutes IS NULL OR p_minutes<1 OR p_minutes>120 THEN
            RAISE_APPLICATION_ERROR(-20120,'Срок резерва должен быть от 1 до 120 минут.');
        END IF;
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        SELECT status,reserved_until INTO l_status,l_until FROM tickets
         WHERE ticket_id=p_ticket_id FOR UPDATE;
        IF l_status='RESERVED' AND l_until<=SYSDATE THEN
            UPDATE tickets SET status='FREE',reserved_by=NULL,reserved_until=NULL
             WHERE ticket_id=p_ticket_id;
            l_status:='FREE';
        END IF;
        IF l_status NOT IN ('FREE','RETURNED') THEN
            RAISE_APPLICATION_ERROR(-20121,'Это место уже занято или зарезервировано.');
        END IF;
        UPDATE tickets SET status='RESERVED',fan_id=NULL,sale_time_id=NULL,
            reserved_by=l_fan_id,reserved_until=SYSDATE+(p_minutes/1440)
         WHERE ticket_id=p_ticket_id;
    EXCEPTION WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20122,'Пользователь или билет не найден.');
    END;

    PROCEDURE purchase_ticket(p_ticket_id IN NUMBER, p_app_user_id IN NUMBER) IS
        l_fan_id NUMBER; l_reserved_by NUMBER; l_until DATE; l_status VARCHAR2(10);
        l_time_id NUMBER(8);
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        SELECT status,reserved_by,reserved_until INTO l_status,l_reserved_by,l_until
          FROM tickets WHERE ticket_id=p_ticket_id FOR UPDATE;
        IF l_status<>'RESERVED' OR l_reserved_by<>l_fan_id OR l_until<=SYSDATE THEN
            RAISE_APPLICATION_ERROR(-20123,'Резерв истёк или билет зарезервирован другим пользователем.');
        END IF;
        SELECT time_id INTO l_time_id FROM dim_time WHERE full_date=TRUNC(SYSDATE);
        UPDATE tickets SET status='SOLD',fan_id=l_fan_id,sale_time_id=l_time_id,
            reserved_by=NULL,reserved_until=NULL WHERE ticket_id=p_ticket_id;
    EXCEPTION WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20124,'Не найден пользователь, билет или календарная дата.');
    END;

    PROCEDURE cancel_reservation(p_ticket_id IN NUMBER, p_app_user_id IN NUMBER) IS
        l_fan_id NUMBER; l_reserved_by NUMBER; l_status VARCHAR2(10);
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        SELECT status,reserved_by INTO l_status,l_reserved_by FROM tickets
         WHERE ticket_id=p_ticket_id FOR UPDATE;
        IF l_status<>'RESERVED' OR l_reserved_by<>l_fan_id THEN
            RAISE_APPLICATION_ERROR(-20125,'Резерв не принадлежит текущему пользователю.');
        END IF;
        UPDATE tickets SET status='FREE',reserved_by=NULL,reserved_until=NULL
         WHERE ticket_id=p_ticket_id;
    EXCEPTION WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20124,'Не найден пользователь или билет.');
    END;

    PROCEDURE my_tickets(p_app_user_id IN NUMBER, p_result OUT SYS_REFCURSOR) IS
        l_fan_id NUMBER;
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        OPEN p_result FOR
            SELECT t.ticket_id,t.status,t.price,t.seat_row,t.seat_number,
                   s.sector_code,m.match_id,m.match_date,m.opponent_name,
                   CASE WHEN t.status='RESERVED' THEN t.reserved_until END AS reserved_until
              FROM tickets t JOIN dim_sector s ON s.sector_id=t.sector_id
              JOIN dim_match m ON m.match_id=t.match_id
             WHERE (t.fan_id=l_fan_id AND t.status IN ('SOLD','RETURNED'))
                OR (t.reserved_by=l_fan_id AND t.status='RESERVED')
             ORDER BY m.match_date DESC,t.ticket_id DESC;
    END;

    PROCEDURE list_favorites(p_app_user_id IN NUMBER, p_result OUT SYS_REFCURSOR) IS
        l_fan_id NUMBER;
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        OPEN p_result FOR
            SELECT p.player_id,p.first_name,p.last_name,p.position,f.created_at
              FROM fan_player_favorites f JOIN dim_player p ON p.player_id=f.player_id
             WHERE f.fan_id=l_fan_id ORDER BY f.created_at DESC;
    END;

    PROCEDURE add_favorite(p_app_user_id IN NUMBER, p_player_id IN NUMBER) IS
        l_fan_id NUMBER;
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        INSERT INTO fan_player_favorites(fan_id,player_id) VALUES(l_fan_id,p_player_id);
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN
        RAISE_APPLICATION_ERROR(-20126,'Игрок уже добавлен в избранное.');
    END;

    PROCEDURE remove_favorite(p_app_user_id IN NUMBER, p_player_id IN NUMBER) IS
        l_fan_id NUMBER;
    BEGIN
        SELECT fan_id INTO l_fan_id FROM app_users
         WHERE user_id=p_app_user_id AND is_active='Y' AND fan_id IS NOT NULL;
        DELETE FROM fan_player_favorites WHERE fan_id=l_fan_id AND player_id=p_player_id;
        IF SQL%ROWCOUNT=0 THEN RAISE_APPLICATION_ERROR(-20127,'Игрок не найден в избранном.'); END IF;
    END;

    PROCEDURE admin_players(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT player_id,first_name,last_name,position,jersey_number,birth_date,
                   country,height_cm,weight_kg,is_active FROM dim_player
             ORDER BY last_name,first_name;
    END;

    PROCEDURE admin_matches(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT m.match_id,m.match_date,m.opponent_name,m.stadium_id,st.name AS stadium,
                   m.is_home,m.match_status,m.goals_for,m.goals_against
              FROM dim_match m JOIN dim_stadium st ON st.stadium_id=m.stadium_id
             ORDER BY m.match_date DESC;
    END;

    PROCEDURE admin_tickets(p_match_id IN NUMBER, p_status IN VARCHAR2,
        p_page IN NUMBER, p_page_size IN NUMBER, p_result OUT SYS_REFCURSOR) IS
        l_page NUMBER:=GREATEST(NVL(p_page,1),1);
        l_size NUMBER:=LEAST(GREATEST(NVL(p_page_size,50),1),200);
    BEGIN
        OPEN p_result FOR
            SELECT * FROM (
                SELECT ROW_NUMBER() OVER(ORDER BY t.ticket_id DESC) rn,
                       t.ticket_id,t.sector_id,m.match_id,m.match_date,m.opponent_name,
                       s.sector_code,t.seat_row,t.seat_number,t.price,t.status,
                       t.reserved_until,t.fan_id,t.reserved_by
                  FROM tickets t JOIN dim_match m ON m.match_id=t.match_id
                  JOIN dim_sector s ON s.sector_id=t.sector_id
                 WHERE (p_match_id IS NULL OR t.match_id=p_match_id)
                   AND (p_status IS NULL OR t.status=UPPER(p_status))
            ) WHERE rn BETWEEN ((l_page-1)*l_size+1) AND (l_page*l_size)
             ORDER BY rn;
    END;

    PROCEDURE admin_merch(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR SELECT merch_id,name,category,price,description,is_active
          FROM dim_merch ORDER BY category,name;
    END;

    PROCEDURE list_fans(p_result OUT SYS_REFCURSOR) IS
    BEGIN

    END;

    PROCEDURE add_ticket(p_match_id IN NUMBER, p_sector_id IN NUMBER,

    PROCEDURE olap_summary(p_date_from IN DATE, p_date_to IN DATE,
        p_sold_count OUT NUMBER, p_revenue OUT NUMBER, p_merch_revenue OUT NUMBER) IS
    BEGIN
        SELECT COUNT(*),NVL(SUM(t.price),0)
          INTO p_sold_count,p_revenue
          FROM tickets t JOIN dim_time d ON d.time_id=t.sale_time_id
         WHERE t.status='SOLD'
           AND (p_date_from IS NULL OR d.full_date>=TRUNC(p_date_from))
           AND (p_date_to IS NULL OR d.full_date<TRUNC(p_date_to)+1);
        SELECT NVL(SUM(o.total_amount),0) INTO p_merch_revenue
          FROM merch_orders o JOIN dim_time d ON d.time_id=o.time_id
         WHERE o.status='COMPLETED'
           AND (p_date_from IS NULL OR d.full_date>=TRUNC(p_date_from))
           AND (p_date_to IS NULL OR d.full_date<TRUNC(p_date_to)+1);
    END;

    PROCEDURE olap_sales_rollup(p_date_from IN DATE, p_date_to IN DATE,
        p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT CASE
                     WHEN GROUPING(d.season)=1 THEN 'ВСЕГО'
                     WHEN GROUPING(d.year_num)=1 THEN 'Сезон '||d.season
                     WHEN GROUPING(d.month_num)=1 THEN TO_CHAR(d.year_num)||' год — всего'
                     ELSE TO_CHAR(d.year_num)||'-'||LPAD(d.month_num,2,'0')||' / '||TRIM(d.month_name)
                   END AS period_label,
                   GROUPING_ID(d.season,d.year_num,d.month_num,d.month_name) AS grouping_level,
                   COUNT(*) AS tickets_sold,NVL(SUM(t.price),0) AS revenue
              FROM tickets t JOIN dim_time d ON d.time_id=t.sale_time_id
             WHERE t.status='SOLD'
               AND (p_date_from IS NULL OR d.full_date>=TRUNC(p_date_from))
               AND (p_date_to IS NULL OR d.full_date<TRUNC(p_date_to)+1)
             GROUP BY ROLLUP(d.season,d.year_num,d.month_num,d.month_name)
             ORDER BY GROUPING_ID(d.season,d.year_num,d.month_num,d.month_name),
                      d.season,d.year_num,d.month_num;
    END;

    PROCEDURE olap_sales_cube(p_date_from IN DATE, p_date_to IN DATE,
        p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT NVL(d.season,'Все сезоны') AS season,
                   NVL(s.category,'Все категории') AS sector_category,
                   GROUPING_ID(d.season,s.category) AS grouping_level,
                   COUNT(*) AS tickets_sold,NVL(SUM(t.price),0) AS revenue
              FROM tickets t JOIN dim_time d ON d.time_id=t.sale_time_id
              JOIN dim_sector s ON s.sector_id=t.sector_id
             WHERE t.status='SOLD'
               AND (p_date_from IS NULL OR d.full_date>=TRUNC(p_date_from))
               AND (p_date_to IS NULL OR d.full_date<TRUNC(p_date_to)+1)
             GROUP BY CUBE(d.season,s.category)
             ORDER BY GROUPING_ID(d.season,s.category),d.season,s.category;
    END;

    PROCEDURE olap_players(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT p.player_id,p.first_name,p.last_name,p.position,
                   NVL(f.fans_count,0) AS fans_count,
                   NVL(s.matches_count,0) AS matches_count,
                   NVL(s.goals_total,0) AS goals_total,
                   NVL(s.assists_total,0) AS assists_total,
                   NVL(s.points_total,0) AS points_total
              FROM dim_player p
              LEFT JOIN (SELECT player_id,COUNT(DISTINCT fan_id) fans_count
                           FROM fan_player_favorites GROUP BY player_id) f
                ON f.player_id=p.player_id
              LEFT JOIN (SELECT player_id,COUNT(*) matches_count,
                                SUM(goals) goals_total,SUM(assists) assists_total,
                                SUM(points) points_total
                           FROM fact_match_stats GROUP BY player_id) s
                ON s.player_id=p.player_id
             ORDER BY NVL(f.fans_count,0) DESC,NVL(s.points_total,0) DESC,p.last_name;
    END;

DECLARE
    l_match_id NUMBER;
    l_sector_a NUMBER;
    l_sector_b NUMBER;
BEGIN
    BEGIN
        SELECT match_id INTO l_match_id FROM dim_match
         WHERE opponent_name=UNISTR('\0414\0435\043C\043E\043D\0441\0442\0440\0430\0446\0438\043E\043D\043D\044B\0439 \0441\043E\043F\0435\0440\043D\0438\043A') AND ROWNUM=1;
        SELECT MIN(sector_id),MAX(sector_id) INTO l_sector_a,l_sector_b
          FROM dim_sector WHERE stadium_id=(SELECT stadium_id FROM dim_match WHERE match_id=l_match_id);
        IF l_sector_a IS NOT NULL AND l_sector_b IS NOT NULL AND l_sector_a<>l_sector_b THEN
            INSERT INTO tickets(match_id,sector_id,seat_row,seat_number,price,status)
            SELECT l_match_id,l_sector_a,CEIL(q.n/100),MOD(q.n-1,100)+1,45,'FREE'
              FROM (SELECT LEVEL n FROM dual CONNECT BY LEVEL<=5000) q
             WHERE NOT EXISTS (SELECT 1 FROM tickets t WHERE t.match_id=l_match_id
                    AND t.sector_id=l_sector_a AND t.seat_row=CEIL(q.n/100)

-- Основание: при диагностике обнаружен матч MATCH_ID=1 на 2026-11-20,
-- арена STADIUM_ID=1 и два сектора A/B по 5 000 мест, но текстовые названия

-- пропустил матч. Скрипт исправляет только этот демонстрационный матч и
-- названия его арены, затем добавляет только отсутствующие места.
-- Он НЕ удаляет строки, не меняет 100 000 тестовых билетов и не пересоздаёт
-- объекты схемы. Повторный запуск безопасен: дубли мест исключены.

DECLARE
    c_demo_match_id CONSTANT NUMBER := 1;
    c_demo_date     CONSTANT DATE := DATE '2026-11-20';
    l_stadium_id    NUMBER;
    l_sector_a      NUMBER;
    l_sector_b      NUMBER;
    l_match_count   NUMBER;
    l_added_a       NUMBER := 0;
    l_added_b       NUMBER := 0;
    l_total         NUMBER;
BEGIN
    -- Не затрагиваем никакой матч, кроме подтверждённого диагностикой MATCH_ID=1.
    SELECT COUNT(*)
      INTO l_match_count
      FROM dim_match
     WHERE match_id = c_demo_match_id
       AND TRUNC(match_date) = c_demo_date
       AND match_status <> 'CANCELLED';

    IF l_match_count <> 1 THEN
        RAISE_APPLICATION_ERROR(-20201,
          'Ожидаемый демонстрационный матч MATCH_ID=1 на 2026-11-20 не найден или неоднозначен. Изменения не внесены.');
    END IF;

    SELECT stadium_id
      INTO l_stadium_id
      FROM dim_match
     WHERE match_id = c_demo_match_id;

    -- Исправляем Unicode-названия, если они были повреждены при первоначальном
    -- запуске SQL*Plus. UNISTR не зависит от кодовой страницы файла/терминала.
    UPDATE dim_stadium
       SET name = UNISTR('\041C\0438\043D\0441\043A-\0410\0440\0435\043D\0430'),
           city = UNISTR('\041C\0438\043D\0441\043A')
     WHERE stadium_id = l_stadium_id;

    UPDATE dim_match
       SET opponent_name = UNISTR('\0414\0435\043C\043E\043D\0441\0442\0440\0430\0446\0438\043E\043D\043D\044B\0439 \0441\043E\043F\0435\0440\043D\0438\043A')

      INTO l_sector_b
      FROM dim_sector
     WHERE stadium_id = l_stadium_id
       AND sector_code = 'B';

    -- билетов сохраняются со своими FREE/RESERVED/SOLD/RETURNED статусами.
    INSERT INTO tickets(match_id, sector_id, seat_row, seat_number, price, status)

              AND t.sector_id = l_sector_a
              AND t.seat_row = CEIL(q.n / 100)
              AND t.seat_number = MOD(q.n - 1, 100) + 1

    INSERT INTO tickets(match_id, sector_id, seat_row, seat_number, price, status)

              AND t.seat_row = CEIL(q.n / 100)
              AND t.seat_number = MOD(q.n - 1, 100) + 1
     );
    l_added_b := SQL%ROWCOUNT;

    SELECT COUNT(*)
      INTO l_total
      FROM tickets
     WHERE match_id = c_demo_match_id

-- =====================================================================
-- Триггеры контроля согласованности и автоматического заполнения дат.
-- =====================================================================
WHENEVER SQLERROR EXIT SQL.SQLCODE

CREATE OR REPLACE TRIGGER trg_dim_match_biu
BEFORE INSERT OR UPDATE OF match_date, time_id ON dim_match
FOR EACH ROW
DECLARE
    l_full_date DATE;
BEGIN
    SELECT full_date INTO l_full_date
      FROM dim_time
     WHERE time_id = :NEW.time_id;
    IF TRUNC(l_full_date) <> TRUNC(:NEW.match_date) THEN
        RAISE_APPLICATION_ERROR(-20001,
            'Дата матча должна совпадать с датой календарного измерения DIM_TIME');
    END IF;
END;
/

CREATE OR REPLACE TRIGGER trg_tickets_biu
BEFORE INSERT OR UPDATE ON tickets
FOR EACH ROW
DECLARE
    l_match_stadium  NUMBER;
    l_sector_stadium NUMBER;
BEGIN
    IF INSERTING THEN
        :NEW.created_at := COALESCE(:NEW.created_at, SYSTIMESTAMP);
        :NEW.updated_at := SYSTIMESTAMP;
    ELSE
        :NEW.created_at := :OLD.created_at;
        :NEW.updated_at := SYSTIMESTAMP;

        IF UPDATING('STATUS') AND :NEW.status <> :OLD.status THEN
            IF NOT (
                (:OLD.status = 'FREE'     AND :NEW.status IN ('RESERVED','SOLD')) OR
                (:OLD.status = 'RESERVED' AND :NEW.status IN ('FREE','SOLD')) OR
                (:OLD.status = 'SOLD'     AND :NEW.status = 'RETURNED') OR
                (:OLD.status = 'RETURNED' AND :NEW.status IN ('RESERVED','SOLD'))
            ) THEN
                RAISE_APPLICATION_ERROR(-20002,
                    'Недопустимый переход статуса билета');
            END IF;
        END IF;
    END IF;

    -- Сектор должен принадлежать тому же стадиону, на котором проводится матч.
    SELECT m.stadium_id, s.stadium_id
      INTO l_match_stadium, l_sector_stadium
      FROM dim_match m CROSS JOIN dim_sector s
     WHERE m.match_id = :NEW.match_id
       AND s.sector_id = :NEW.sector_id;

    IF l_match_stadium <> l_sector_stadium THEN
        RAISE_APPLICATION_ERROR(-20003,
            'Выбранный сектор не относится к стадиону указанного матча');
    END IF;
END;
/

WHENEVER SQLERROR EXIT SQL.SQLCODE

CREATE OR REPLACE FUNCTION f_ticket_revenue(
    p_date_from IN DATE,
    p_date_to   IN DATE
) RETURN NUMBER
AUTHID DEFINER
IS
    l_revenue NUMBER(14,2);
BEGIN
    IF p_date_from IS NULL OR p_date_to IS NULL OR TRUNC(p_date_from) > TRUNC(p_date_to) THEN
        RAISE_APPLICATION_ERROR(-20010, 'Некорректный диапазон дат для расчёта выручки');
    END IF;

    SELECT NVL(SUM(t.price), 0)
      INTO l_revenue
      FROM tickets t
      JOIN dim_time d ON d.time_id = t.sale_time_id
     WHERE t.status = 'SOLD'
       AND d.full_date >= TRUNC(p_date_from)
       AND d.full_date < TRUNC(p_date_to) + 1;

    RETURN l_revenue;
END;
/

CREATE OR REPLACE FUNCTION f_player_popularity(
    p_player_id IN NUMBER
) RETURN NUMBER
AUTHID DEFINER
IS
    l_count NUMBER;
BEGIN
    SELECT COUNT(DISTINCT fan_id)
      INTO l_count
      FROM fan_player_favorites
     WHERE player_id = p_player_id;
    RETURN l_count;
END;

CREATE OR REPLACE VIEW v_ticket_sales_by_period AS
SELECT d.full_date,
       d.year_num,
       d.month_num,
       d.month_name,
       d.season,
       COUNT(*) AS tickets_sold,
       SUM(t.price) AS revenue
  FROM tickets t
  JOIN dim_time d ON d.time_id = t.sale_time_id
 WHERE t.status = 'SOLD'
 GROUP BY d.full_date, d.year_num, d.month_num, d.month_name, d.season;

CREATE OR REPLACE VIEW v_player_popularity AS
SELECT p.player_id,
       p.first_name,
       p.last_name,
       COUNT(DISTINCT f.fan_id) AS fans_favorited
  FROM dim_player p
  LEFT JOIN fan_player_favorites f ON f.player_id = p.player_id
 GROUP BY p.player_id, p.first_name, p.last_name;

CREATE OR REPLACE VIEW v_player_performance AS
SELECT p.player_id,
       p.first_name,
       p.last_name,
       COUNT(s.stat_id) AS matches_count,
       NVL(SUM(s.goals), 0) AS goals_total,
       NVL(SUM(s.assists), 0) AS assists_total,
       NVL(SUM(s.points), 0) AS points_total,
       NVL(SUM(s.shots), 0) AS shots_total,
       NVL(SUM(s.penalty_minutes), 0) AS penalty_minutes_total
  FROM dim_player p
  LEFT JOIN fact_match_stats s ON s.player_id = p.player_id
 GROUP BY p.player_id, p.first_name, p.last_name;

CREATE OR REPLACE VIEW v_merch_sales_by_period AS
SELECT d.full_date,
       d.year_num,
       d.month_num,
       d.season,
       m.merch_id,
       m.name AS merch_name,
       SUM(o.quantity) AS quantity_sold,
       SUM(o.total_amount) AS revenue
  FROM merch_orders o
  JOIN dim_time d ON d.time_id = o.time_id
  JOIN dim_merch m ON m.merch_id = o.merch_id
 WHERE o.status = 'COMPLETED'
 GROUP BY d.full_date, d.year_num, d.month_num, d.season, m.merch_id, m.name;

CREATE OR REPLACE PACKAGE BODY pkg_admin_api AS
    PROCEDURE add_player(
        p_first_name IN VARCHAR2, p_last_name IN VARCHAR2,
        p_position IN VARCHAR2, p_jersey_number IN NUMBER,
        p_birth_date IN DATE, p_country IN VARCHAR2,
        p_height_cm IN NUMBER, p_weight_kg IN NUMBER,
        p_player_id OUT NUMBER) IS
    BEGIN
        INSERT INTO dim_player(first_name,last_name,position,jersey_number,
                               birth_date,country,height_cm,weight_kg)
        VALUES (TRIM(p_first_name),TRIM(p_last_name),UPPER(TRIM(p_position)),
                p_jersey_number,p_birth_date,p_country,p_height_cm,p_weight_kg)
        RETURNING player_id INTO p_player_id;
    END;

    PROCEDURE update_player(
        p_player_id IN NUMBER, p_first_name IN VARCHAR2,
        p_last_name IN VARCHAR2, p_position IN VARCHAR2,
        p_jersey_number IN NUMBER, p_birth_date IN DATE,
        p_country IN VARCHAR2, p_height_cm IN NUMBER,
        p_weight_kg IN NUMBER, p_is_active IN CHAR) IS
    BEGIN
        UPDATE dim_player
           SET first_name=TRIM(p_first_name), last_name=TRIM(p_last_name),
               position=UPPER(TRIM(p_position)), jersey_number=p_jersey_number,
               birth_date=p_birth_date, country=p_country,
               height_cm=p_height_cm, weight_kg=p_weight_kg,
               is_active=UPPER(p_is_active)
         WHERE player_id=p_player_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20020,'Игрок не найден');
        END IF;
    END;

    PROCEDURE delete_player(p_player_id IN NUMBER) IS
    BEGIN
        DELETE FROM dim_player WHERE player_id=p_player_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20021,'Игрок не найден');
        END IF;
        -- Если игрок уже присутствует в статистике или избранном,
        -- Oracle запретит удаление по внешнему ключу. В таком случае
        -- рекомендуется использовать UPDATE is_active='N'.
    END;

    PROCEDURE add_match(
        p_match_date IN DATE, p_opponent_name IN VARCHAR2,
        p_stadium_id IN NUMBER, p_is_home IN CHAR,
        p_match_id OUT NUMBER) IS
        l_time_id NUMBER(8);
    BEGIN
        SELECT time_id INTO l_time_id
          FROM dim_time
         WHERE full_date=TRUNC(p_match_date);
        INSERT INTO dim_match(match_date,time_id,opponent_name,stadium_id,is_home)
        VALUES(p_match_date,l_time_id,TRIM(p_opponent_name),p_stadium_id,UPPER(p_is_home))
        RETURNING match_id INTO p_match_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20022,'Для даты матча отсутствует строка в DIM_TIME');
    END;

    PROCEDURE update_match(
        p_match_id IN NUMBER, p_match_date IN DATE,
        p_opponent_name IN VARCHAR2, p_stadium_id IN NUMBER,
        p_is_home IN CHAR, p_match_status IN VARCHAR2,
        p_goals_for IN NUMBER, p_goals_against IN NUMBER) IS
        l_time_id NUMBER(8);
    BEGIN
        SELECT time_id INTO l_time_id
          FROM dim_time
         WHERE full_date=TRUNC(p_match_date);
        UPDATE dim_match
           SET match_date=p_match_date, time_id=l_time_id,
               opponent_name=TRIM(p_opponent_name), stadium_id=p_stadium_id,
               is_home=UPPER(p_is_home), match_status=UPPER(p_match_status),
               goals_for=p_goals_for, goals_against=p_goals_against
         WHERE match_id=p_match_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20023,'Матч не найден');
        END IF;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20022,'Для даты матча отсутствует строка в DIM_TIME');
    END;

    PROCEDURE delete_match(p_match_id IN NUMBER) IS
    BEGIN
        DELETE FROM dim_match WHERE match_id=p_match_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20024,'Матч не найден');
        END IF;
    END;

    PROCEDURE add_ticket(
        p_match_id IN NUMBER, p_sector_id IN NUMBER,
        p_seat_row IN NUMBER, p_seat_number IN NUMBER,
        p_price IN NUMBER, p_ticket_id OUT NUMBER) IS
    BEGIN
        INSERT INTO tickets(match_id,sector_id,seat_row,seat_number,price,status)
        VALUES(p_match_id,p_sector_id,p_seat_row,p_seat_number,p_price,'FREE')
        RETURNING ticket_id INTO p_ticket_id;
    END;

    PROCEDURE sell_ticket(p_ticket_id IN NUMBER, p_fan_id IN NUMBER) IS
        l_status VARCHAR2(10);
        l_time_id NUMBER(8);
    BEGIN
        SELECT status INTO l_status
          FROM tickets
         WHERE ticket_id=p_ticket_id
           FOR UPDATE;
        IF l_status NOT IN ('FREE','RESERVED','RETURNED') THEN
            RAISE_APPLICATION_ERROR(-20025,'Билет нельзя продать в текущем статусе');
        END IF;
        SELECT time_id INTO l_time_id
          FROM dim_time
         WHERE full_date=TRUNC(SYSDATE);
        UPDATE tickets
           SET status='SOLD', fan_id=p_fan_id, sale_time_id=l_time_id,
               reserved_by=NULL, reserved_until=NULL
         WHERE ticket_id=p_ticket_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20026,'Билет не найден или текущая дата отсутствует в DIM_TIME');
    END;

    PROCEDURE return_ticket(p_ticket_id IN NUMBER) IS
        l_status VARCHAR2(10);
    BEGIN
        SELECT status INTO l_status
          FROM tickets
         WHERE ticket_id=p_ticket_id
           FOR UPDATE;
        IF l_status <> 'SOLD' THEN
            RAISE_APPLICATION_ERROR(-20027,'Возврат возможен только для проданного билета');
        END IF;
        UPDATE tickets
           SET status='RETURNED', reserved_by=NULL, reserved_until=NULL
         WHERE ticket_id=p_ticket_id;
        -- fan_id и sale_time_id сохраняются как сведения о состоявшейся продаже.
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20026,'Билет не найден');
    END;

    PROCEDURE release_expired_reservations(p_released_count OUT NUMBER) IS
    BEGIN
        UPDATE tickets
           SET status='FREE', reserved_by=NULL, reserved_until=NULL
         WHERE status='RESERVED' AND reserved_until <= SYSDATE;
        p_released_count := SQL%ROWCOUNT;
    END;

    PROCEDURE add_merch(
        p_name IN VARCHAR2, p_category IN VARCHAR2,
        p_price IN NUMBER, p_description IN VARCHAR2,
        p_merch_id OUT NUMBER) IS
    BEGIN
        INSERT INTO dim_merch(name,category,price,description)
        VALUES(TRIM(p_name),TRIM(p_category),p_price,p_description)
        RETURNING merch_id INTO p_merch_id;
    END;

        p_merch_id IN NUMBER, p_name IN VARCHAR2,
        p_category IN VARCHAR2, p_price IN NUMBER,
        p_description IN VARCHAR2, p_is_active IN CHAR) IS
    BEGIN
        UPDATE dim_merch
           SET name=TRIM(p_name), category=TRIM(p_category), price=p_price,
               description=p_description, is_active=UPPER(p_is_active)
         WHERE merch_id=p_merch_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20028,'Товар не найден');
        END IF;
    END;

    PROCEDURE delete_merch(p_merch_id IN NUMBER) IS
    BEGIN
        DELETE FROM dim_merch WHERE merch_id=p_merch_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20029,'Товар не найден');
        END IF;
    END;

    PROCEDURE add_merch_order(
        p_merch_id IN NUMBER, p_fan_id IN NUMBER,
        p_time_id IN NUMBER, p_quantity IN NUMBER,
        p_order_id OUT NUMBER) IS
        l_price NUMBER(10,2);
    BEGIN
        SELECT price INTO l_price
          FROM dim_merch
         WHERE merch_id=p_merch_id AND is_active='Y';
        INSERT INTO merch_orders(merch_id,fan_id,time_id,quantity,unit_price,status)
        VALUES(p_merch_id,p_fan_id,p_time_id,p_quantity,l_price,'COMPLETED')
        RETURNING order_id INTO p_order_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20030,'Активный товар не найден');

               AND (p_date_to IS NULL OR full_date < TRUNC(p_date_to)+1)

    END;

    END;
END pkg_admin_api;

CREATE OR REPLACE PACKAGE BODY pkg_user_api AS
    FUNCTION current_fan_id(p_app_user_id IN NUMBER) RETURN NUMBER IS
        l_fan_id NUMBER;
    BEGIN
        SELECT fan_id INTO l_fan_id
          FROM app_users
         WHERE user_id=p_app_user_id
           AND is_active='Y'
           AND role='USER'
           AND fan_id IS NOT NULL
           AND UPPER(db_username)=SYS_CONTEXT('USERENV','SESSION_USER');
        RETURN l_fan_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20040,
              'Для текущего Oracle-сеанса не найдена активная APP_USERS-запись USER с привязанным DIM_FAN');
    END;

    PROCEDURE list_players(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT player_id, first_name, last_name, position, jersey_number,
                   birth_date, country, is_active
              FROM dim_player
             WHERE is_active='Y'
             ORDER BY last_name, first_name;
    END;

    PROCEDURE list_matches(p_date_from IN DATE, p_date_to IN DATE,
                           p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT m.match_id, m.match_date, m.opponent_name, s.name AS stadium,
                   m.is_home, m.goals_for, m.goals_against, m.match_status
              FROM dim_match m
              JOIN dim_stadium s ON s.stadium_id=m.stadium_id
             WHERE (p_date_from IS NULL OR m.match_date >= TRUNC(p_date_from))
               AND (p_date_to IS NULL OR m.match_date < TRUNC(p_date_to)+1)
             ORDER BY m.match_date;
    END;

    PROCEDURE list_available_tickets(p_match_id IN NUMBER,
                                     p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT t.ticket_id, t.match_id, s.sector_code, t.seat_row,
                   t.seat_number, t.price, t.status
              FROM tickets t
              JOIN dim_sector s ON s.sector_id=t.sector_id
             WHERE t.match_id=p_match_id
               AND (t.status='FREE' OR
                    (t.status='RESERVED' AND t.reserved_until <= SYSDATE))
             ORDER BY s.sector_code, t.seat_row, t.seat_number;
    END;

    PROCEDURE list_merch(p_result OUT SYS_REFCURSOR) IS
    BEGIN
        OPEN p_result FOR
            SELECT merch_id, name, category, price, description
              FROM dim_merch
             WHERE is_active='Y'
             ORDER BY category, name;
    END;

    PROCEDURE reserve_ticket(p_ticket_id IN NUMBER, p_app_user_id IN NUMBER,
                             p_minutes IN NUMBER DEFAULT 15) IS
        l_fan_id NUMBER;
        l_status VARCHAR2(10);
        l_reserved_until DATE;
    BEGIN
        IF p_minutes IS NULL OR p_minutes < 1 OR p_minutes > 120 THEN
            RAISE_APPLICATION_ERROR(-20042,'Срок резерва должен составлять от 1 до 120 минут');
        END IF;
        l_fan_id := current_fan_id(p_app_user_id);
        SELECT status, reserved_until
          INTO l_status, l_reserved_until
          FROM tickets
         WHERE ticket_id=p_ticket_id
           FOR UPDATE;

        IF l_status='RESERVED' AND l_reserved_until <= SYSDATE THEN
            UPDATE tickets
               SET status='FREE', reserved_by=NULL, reserved_until=NULL
             WHERE ticket_id=p_ticket_id;
            l_status := 'FREE';
        END IF;

        IF l_status NOT IN ('FREE','RETURNED') THEN
            RAISE_APPLICATION_ERROR(-20043,'Билет уже продан или зарезервирован');
        END IF;

        UPDATE tickets
           SET status='RESERVED', fan_id=NULL, sale_time_id=NULL,
               reserved_by=l_fan_id, reserved_until=SYSDATE+(p_minutes/1440)
         WHERE ticket_id=p_ticket_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20044,'Билет не найден');
    END;

    PROCEDURE add_favorite(p_app_user_id IN NUMBER, p_player_id IN NUMBER) IS
        l_fan_id NUMBER;
    BEGIN
        l_fan_id := current_fan_id(p_app_user_id);
        INSERT INTO fan_player_favorites(fan_id,player_id)
        VALUES(l_fan_id,p_player_id);
    EXCEPTION
        WHEN DUP_VAL_ON_INDEX THEN
            RAISE_APPLICATION_ERROR(-20045,'Игрок уже находится в списке избранного');
    END;

    PROCEDURE remove_favorite(p_app_user_id IN NUMBER, p_player_id IN NUMBER) IS
        l_fan_id NUMBER;
    BEGIN
        l_fan_id := current_fan_id(p_app_user_id);
        DELETE FROM fan_player_favorites
         WHERE fan_id=l_fan_id AND player_id=p_player_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20046,'Игрок не найден в списке избранного');
        END IF;
    END;

    PROCEDURE list_favorites(p_app_user_id IN NUMBER, p_result OUT SYS_REFCURSOR) IS
        l_fan_id NUMBER;
    BEGIN
        l_fan_id := current_fan_id(p_app_user_id);
        OPEN p_result FOR
            SELECT p.player_id,p.first_name,p.last_name,p.position,f.created_at
              FROM fan_player_favorites f
              JOIN dim_player p ON p.player_id=f.player_id
             WHERE f.fan_id=l_fan_id
             ORDER BY f.created_at DESC;
    END;

END pkg_user_api;

CREATE OR REPLACE PACKAGE BODY pkg_security_api AS
    PROCEDURE create_app_user(
        p_username IN VARCHAR2, p_password_hash IN VARCHAR2,
        p_email IN VARCHAR2, p_role IN VARCHAR2,
        p_fan_id IN NUMBER, p_user_id OUT NUMBER) IS
        l_role VARCHAR2(20);
    BEGIN
        l_role := UPPER(TRIM(p_role));
        IF l_role NOT IN ('MAIN_ADMIN','ADMIN','USER') THEN
            RAISE_APPLICATION_ERROR(-20050,'Недопустимая прикладная роль');
        END IF;
        IF p_password_hash IS NULL OR LENGTH(TRIM(p_password_hash)) < 20 THEN
            RAISE_APPLICATION_ERROR(-20051,'Передайте хеш пароля, а не открытый пароль');
        END IF;
        INSERT INTO app_users(username,password_hash,email,role,fan_id)
        VALUES(TRIM(p_username),p_password_hash,LOWER(TRIM(p_email)),l_role,p_fan_id)
        RETURNING user_id INTO p_user_id;
    END;

    PROCEDURE change_app_user_role(p_user_id IN NUMBER, p_role IN VARCHAR2) IS
        l_role VARCHAR2(20);
    BEGIN
        l_role := UPPER(TRIM(p_role));
        IF l_role NOT IN ('MAIN_ADMIN','ADMIN','USER') THEN
            RAISE_APPLICATION_ERROR(-20050,'Недопустимая прикладная роль');
        END IF;
        UPDATE app_users SET role=l_role WHERE user_id=p_user_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20052,'Прикладная учётная запись не найдена');
        END IF;
    END;

    PROCEDURE set_app_user_active(p_user_id IN NUMBER, p_is_active IN CHAR) IS
    BEGIN
        IF UPPER(p_is_active) NOT IN ('Y','N') THEN
            RAISE_APPLICATION_ERROR(-20053,'Признак активности должен быть Y или N');
        END IF;
        UPDATE app_users SET is_active=UPPER(p_is_active) WHERE user_id=p_user_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20052,'Прикладная учётная запись не найдена');
        END IF;
    END;

    PROCEDURE bind_app_user_db_user(p_user_id IN NUMBER, p_db_username IN VARCHAR2) IS
        l_count PLS_INTEGER;
    BEGIN
        IF p_db_username IS NULL OR TRIM(p_db_username) IS NULL THEN
            RAISE_APPLICATION_ERROR(-20054,'Необходимо указать Oracle-пользователя');
        END IF;
        SELECT COUNT(*) INTO l_count
          FROM all_users
         WHERE username=UPPER(TRIM(p_db_username));
        IF l_count=0 THEN
            RAISE_APPLICATION_ERROR(-20055,'Указанный Oracle-пользователь не существует');
        END IF;
        UPDATE app_users
           SET db_username=UPPER(TRIM(p_db_username))
         WHERE user_id=p_user_id;
        IF SQL%ROWCOUNT=0 THEN
            RAISE_APPLICATION_ERROR(-20052,'Прикладная учётная запись не найдена');
        END IF;
    END;
END pkg_security_api;

CREATE OR REPLACE PACKAGE BODY pkg_json_io AS
    PROCEDURE read_file_to_clob(p_file_name IN VARCHAR2, p_json OUT CLOB) IS
        l_file UTL_FILE.FILE_TYPE;
        l_line VARCHAR2(32767);
    BEGIN
        DBMS_LOB.CREATETEMPORARY(p_json, TRUE);
        l_file := UTL_FILE.FOPEN('HK_JSON_DIR',p_file_name,'R',32767);
        LOOP
            BEGIN
                UTL_FILE.GET_LINE(l_file,l_line);
                DBMS_LOB.WRITEAPPEND(p_json,LENGTH(l_line),l_line);
                DBMS_LOB.WRITEAPPEND(p_json,1,CHR(10));
            EXCEPTION
                WHEN NO_DATA_FOUND THEN EXIT;
            END;
        END LOOP;
        UTL_FILE.FCLOSE(l_file);
    EXCEPTION
        WHEN OTHERS THEN
            IF UTL_FILE.IS_OPEN(l_file) THEN UTL_FILE.FCLOSE(l_file); END IF;
            IF p_json IS NOT NULL AND DBMS_LOB.ISTEMPORARY(p_json)=1 THEN
                DBMS_LOB.FREETEMPORARY(p_json);
            END IF;
            RAISE;
    END;

    PROCEDURE import_players(p_file_name IN VARCHAR2) IS
        l_json CLOB;
    BEGIN
        read_file_to_clob(p_file_name,l_json);
        INSERT INTO dim_player(first_name,last_name,position,jersey_number,
                               birth_date,country,height_cm,weight_kg)
        SELECT jt.first_name,
               jt.last_name,
               UPPER(jt.position),
               jt.jersey_number,
               CASE WHEN jt.birth_date IS NOT NULL
                    THEN TO_DATE(jt.birth_date,'YYYY-MM-DD') END,
               jt.country,
               jt.height_cm,
               jt.weight_kg
          FROM JSON_TABLE(l_json,'$.players[*]'
                 COLUMNS (
                   first_name    VARCHAR2(50) PATH '$.first_name',
                   last_name     VARCHAR2(50) PATH '$.last_name',
                   position      VARCHAR2(15) PATH '$.position',
                   jersey_number NUMBER       PATH '$.jersey_number' NULL ON ERROR,
                   birth_date    VARCHAR2(10) PATH '$.birth_date' NULL ON ERROR,
                   country       VARCHAR2(50) PATH '$.country' NULL ON ERROR,
                   height_cm     NUMBER       PATH '$.height_cm' NULL ON ERROR,
                   weight_kg     NUMBER       PATH '$.weight_kg' NULL ON ERROR
                 )) jt;
        DBMS_LOB.FREETEMPORARY(l_json);
    EXCEPTION
        WHEN OTHERS THEN
            IF l_json IS NOT NULL AND DBMS_LOB.ISTEMPORARY(l_json)=1 THEN
                DBMS_LOB.FREETEMPORARY(l_json);
            END IF;
            RAISE;
    END;

    PROCEDURE export_players(p_file_name IN VARCHAR2) IS
        l_json CLOB;
        l_file UTL_FILE.FILE_TYPE;
        l_offset PLS_INTEGER := 1;
        l_chunk VARCHAR2(32767);
        l_length PLS_INTEGER;
    BEGIN
        SELECT JSON_OBJECT(
                 'players' VALUE (
                   SELECT JSON_ARRAYAGG(
                            JSON_OBJECT(
                              'player_id' VALUE player_id,
                              'first_name' VALUE first_name,
                              'last_name' VALUE last_name,
                              'position' VALUE position,
                              'jersey_number' VALUE jersey_number,
                              'birth_date' VALUE CASE WHEN birth_date IS NOT NULL
                                                       THEN TO_CHAR(birth_date,'YYYY-MM-DD') END,
                              'country' VALUE country,
                              'height_cm' VALUE height_cm,
                              'weight_kg' VALUE weight_kg,
                              'is_active' VALUE is_active
                              NULL ON NULL RETURNING CLOB
                            ) RETURNING CLOB
                          )
                     FROM dim_player
                 ) FORMAT JSON
                 RETURNING CLOB
               )
          INTO l_json
          FROM dual;

        l_file := UTL_FILE.FOPEN('HK_JSON_DIR',p_file_name,'W',32767);
        l_length := DBMS_LOB.GETLENGTH(l_json);
        WHILE l_offset <= l_length LOOP
            l_chunk := DBMS_LOB.SUBSTR(l_json,32767,l_offset);
            UTL_FILE.PUT(l_file,l_chunk);
            l_offset := l_offset + LENGTH(l_chunk);
        END LOOP;
        UTL_FILE.NEW_LINE(l_file);
        UTL_FILE.FCLOSE(l_file);
        DBMS_LOB.FREETEMPORARY(l_json);
    EXCEPTION
        WHEN OTHERS THEN
            IF UTL_FILE.IS_OPEN(l_file) THEN UTL_FILE.FCLOSE(l_file); END IF;
            IF l_json IS NOT NULL AND DBMS_LOB.ISTEMPORARY(l_json)=1 THEN
                DBMS_LOB.FREETEMPORARY(l_json);
            END IF;
            RAISE;
    END;
END pkg_json_io;

-- =====================================================================
-- Загрузка минимум 100 000 тестовых строк в TICKETS.
-- Выполнять от HK_DINAMO после seed_demo_data.sql.
-- Десять тестовых матчей, по 10 000 мест на матч (2 сектора по 5 000).
-- =====================================================================
WHENEVER SQLERROR EXIT SQL.SQLCODE
SET SERVEROUTPUT ON

DECLARE
    l_stadium_id NUMBER;
    l_sector_a   NUMBER;
    l_sector_b   NUMBER;
    l_fan_id     NUMBER;
    l_match_id   NUMBER;
    l_time_id    NUMBER(8);
    l_existing   NUMBER;
    l_total      NUMBER;
BEGIN
    SELECT COUNT(*) INTO l_existing
      FROM dim_match
     WHERE opponent_name LIKE 'TEST PERF %';
    IF l_existing > 0 THEN
        RAISE_APPLICATION_ERROR(-20100,
          'Тестовые матчи уже существуют. Для повторного запуска очистите тестовые матчи и билеты либо создайте новую схему.');
    END IF;

    SELECT MIN(stadium_id) INTO l_stadium_id FROM dim_stadium;
    SELECT MIN(sector_id), MAX(sector_id)
      INTO l_sector_a, l_sector_b
      FROM dim_sector
     WHERE stadium_id=l_stadium_id;
    SELECT MIN(fan_id) INTO l_fan_id FROM dim_fan;

    IF l_stadium_id IS NULL OR l_sector_a IS NULL OR l_sector_b IS NULL OR l_sector_a=l_sector_b OR l_fan_id IS NULL THEN
        RAISE_APPLICATION_ERROR(-20101,'Сначала выполните загрузку демонстрационных данных');
    END IF;

    FOR i IN 1..10 LOOP
        SELECT time_id INTO l_time_id
          FROM dim_time
         WHERE full_date=TRUNC(SYSDATE)+i;

        INSERT INTO dim_match(match_date,time_id,opponent_name,stadium_id,is_home)
        VALUES(TRUNC(SYSDATE)+i,l_time_id,'TEST PERF '||TO_CHAR(i,'FM00'),l_stadium_id,'Y')
        RETURNING match_id INTO l_match_id;

        INSERT INTO tickets(match_id,sector_id,seat_row,seat_number,price,status,
                             fan_id,sale_time_id,reserved_by,reserved_until)
        SELECT l_match_id,
               CASE WHEN q.n <= 5000 THEN l_sector_a ELSE l_sector_b END,
               CEIL(q.seat_pos/100),
               MOD(q.seat_pos-1,100)+1,
               40 + MOD(q.n,5)*5,
               CASE MOD(q.n,4)
                    WHEN 0 THEN 'FREE'
                    WHEN 1 THEN 'RESERVED'
                    WHEN 2 THEN 'SOLD'
                    ELSE 'RETURNED'
               END,
               CASE WHEN MOD(q.n,4)=2 THEN l_fan_id END,
               CASE WHEN MOD(q.n,4)=2 THEN l_time_id END,
               CASE WHEN MOD(q.n,4)=1 THEN l_fan_id END,
               CASE WHEN MOD(q.n,4)=1 THEN SYSDATE+1 END
          FROM (SELECT LEVEL AS n,
                       CASE WHEN LEVEL<=5000 THEN LEVEL ELSE LEVEL-5000 END AS seat_pos
                  FROM dual
                CONNECT BY LEVEL<=10000) q;
    END LOOP;

    SELECT COUNT(*) INTO l_total
      FROM tickets t
      JOIN dim_match m ON m.match_id=t.match_id
     WHERE m.opponent_name LIKE 'TEST PERF %';

    IF l_total < 100000 THEN
        RAISE_APPLICATION_ERROR(-20102,'Не удалось загрузить требуемые 100 000 строк');
    END IF;

    DBMS_STATS.GATHER_TABLE_STATS(USER,'TICKETS',cascade=>TRUE);
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Загружено строк в TICKETS для тестирования: '||l_total);
END;
